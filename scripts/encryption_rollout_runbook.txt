STEP 3 - SQL - Preview which sales leads would be linked to a client (changes nothing)
Pass: a table appears (it may be empty). Save a screenshot of it - it is the undo list.
If ANY row says 'will link', do step 4. If there are no rows at all, skip step 4.

----- copy from here -----
select c.id as client_id,
       min(l.id) as lead_row_id,
       count(l.id) as candidate_leads,
       case when count(l.id) = 1 then 'will link'
            else 'AMBIGUOUS - link by hand' end as action
from clients c
join leads l
  on lower(l.client_email) = lower(c.email)
 and l.client_id is null
 and (l.answers ->> '_upgrade_request') is null
where c.email is not null and c.email <> ''
  and not exists (select 1 from leads x where x.client_id = c.id
                  and (x.answers ->> '_upgrade_request') is null)
group by c.id
order by c.id;

STEP 4 - SQL - Link the 'will link' leads (ONLY if step 3 showed a 'will link' row)
Pass: "Success" with a number of rows equal to the 'will link' rows from step 3.
----- copy from here -----
with candidates as (
  select c.id as client_id, min(l.id) as lead_row_id, count(l.id) as n
  from clients c
  join leads l
    on lower(l.client_email) = lower(c.email)
   and l.client_id is null
   and (l.answers ->> '_upgrade_request') is null
  where c.email is not null and c.email <> ''
    and not exists (select 1 from leads x where x.client_id = c.id
                    and (x.answers ->> '_upgrade_request') is null)
  group by c.id
)
update leads l
   set client_id = cand.client_id
  from candidates cand
 where l.id = cand.lead_row_id
   and cand.n = 1
   and l.client_id is null;
